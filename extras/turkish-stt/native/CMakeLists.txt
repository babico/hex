cmake_minimum_required(VERSION 3.20)
project(hex_sherpa_worker LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SHERPA_ROOT "" CACHE PATH "Extracted official sherpa-onnx 1.13.8 shared SDK")
find_path(SHERPA_INCLUDE sherpa-onnx/c-api/cxx-api.h PATHS "${SHERPA_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
find_library(SHERPA_CXX NAMES sherpa-onnx-cxx-api PATHS "${SHERPA_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
add_executable(hex-sherpa-worker sherpa_worker.cpp)
target_include_directories(hex-sherpa-worker PRIVATE "${SHERPA_INCLUDE}")
target_link_libraries(hex-sherpa-worker PRIVATE "${SHERPA_CXX}")
if(APPLE)
  set_target_properties(hex-sherpa-worker PROPERTIES INSTALL_RPATH "@loader_path")
elseif(UNIX)
  set_target_properties(hex-sherpa-worker PROPERTIES INSTALL_RPATH "$ORIGIN")
endif()
install(TARGETS hex-sherpa-worker RUNTIME DESTINATION .)
install(DIRECTORY "${SHERPA_ROOT}/lib/" DESTINATION . FILES_MATCHING
  PATTERN "*.so*" PATTERN "*.dylib" PATTERN "*.dll")
if(IS_DIRECTORY "${SHERPA_ROOT}/bin")
  install(DIRECTORY "${SHERPA_ROOT}/bin/" DESTINATION . FILES_MATCHING PATTERN "*.dll")
endif()
