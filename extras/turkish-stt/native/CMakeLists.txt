cmake_minimum_required(VERSION 3.20)
project(hex_sherpa_worker LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SHERPA_ROOT "" CACHE PATH "Extracted checksum-verified sherpa-onnx 1.13.8 SDK")
find_library(SHERPA_C NAMES sherpa-onnx-c-api PATHS "${SHERPA_ROOT}/lib" "${SHERPA_ROOT}/bin" NO_DEFAULT_PATH REQUIRED)
add_executable(hex-sherpa-worker sherpa_worker.cpp)
target_include_directories(hex-sherpa-worker PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/vendor")
target_compile_definitions(hex-sherpa-worker PRIVATE SHERPA_ONNX_BUILD_SHARED_LIBS=1)
target_link_libraries(hex-sherpa-worker PRIVATE "${SHERPA_C}")
if(APPLE)
  set_target_properties(hex-sherpa-worker PROPERTIES INSTALL_RPATH "@loader_path")
elseif(UNIX)
  set_target_properties(hex-sherpa-worker PROPERTIES INSTALL_RPATH "$ORIGIN")
  target_link_options(hex-sherpa-worker PRIVATE "-Wl,-rpath-link,${SHERPA_ROOT}/lib")
endif()
install(TARGETS hex-sherpa-worker RUNTIME DESTINATION .)
install(DIRECTORY "${SHERPA_ROOT}/lib/" DESTINATION . FILES_MATCHING
  PATTERN "*.so*" PATTERN "*.dylib" PATTERN "*.dll")
if(IS_DIRECTORY "${SHERPA_ROOT}/bin")
  install(DIRECTORY "${SHERPA_ROOT}/bin/" DESTINATION . FILES_MATCHING PATTERN "*.dll")
endif()
install(FILES vendor/LICENSE DESTINATION sdk-notices RENAME sherpa-onnx-LICENSE)
